    def generate_projected_metrics(self, prev: Metrics, summ: CohortSummary, delta: TouchpointDelta) -> Metrics:
        friction = delta.friction_coefficient
        verify_penalty = 1.0 if delta.randomize_verification else 0.0
        
        abandon = clamp01(prev.abandon_rate + (friction * 0.15) - (summ.avg_abandonment_threshold * 0.05))
        reentry = clamp01(prev.ivr_reentry_rate + (friction * 0.10) + (verify_penalty * 0.05))
        
        transfer = inv_logit(
            logit(max(prev.transfer_rate, 0.01))
            + summ.avg_escalation_propensity * 1.3
            + friction * 0.5
            + prev.repeat_auth_rate * 0.6
            + (1.0 - prev.determinism_index) * 0.8
        )
        
        callback = inv_logit(
            logit(max(prev.callback_rate, 0.05))
            + friction * 0.45
            + prev.short_disconnect_rate * 0.8
            + abandon * 0.5
        )
        
        backoffice = inv_logit(
            logit(max(prev.backoffice_trigger, 0.03))
            + transfer * 0.6
            + prev.repeat_auth_rate * 0.35
        )
        
        determinism = clamp01(
            prev.determinism_index
            - friction * 0.10
            - verify_penalty * 0.20
            - reentry * 0.04
        )
        
        containment = clamp01(1.0 - safe_rate_sum(abandon, transfer, callback * 0.30))
        
        return Metrics(
            abandon_rate=abandon,
            ivr_reentry_rate=reentry,
            transfer_rate=transfer,
            aht=prev.aht,
            callback_rate=callback,
            backoffice_trigger=backoffice,
            determinism_index=determinism,
            repeat_auth_rate=prev.repeat_auth_rate,
            containment=containment,
            short_disconnect_rate=prev.short_disconnect_rate,
        )# (PredictiveReactionSandboxLogOdds implementation omitted for brevity, # following same class structure as above)

    def __init__(self, baseline: Metrics):
        self.baseline = baseline

    def feedback(self, prev: Metrics, current: Metrics) -> Metrics:
        blended_abandon = clamp01(0.35 * prev.abandon_rate + 0.75 * current.abandon_rate)
        blended_reentry = clamp01(0.30 * prev.ivr_reentry_rate + 0.80 * current.ivr_reentry_rate)
        
        instability_push = (0.35 * current.abandon_rate + 0.25 * current.ivr_reentry_rate + 0.20 * current.repeat_auth_rate)
        
        transfer = clamp01(0.25 * prev.transfer_rate + 0.75 * current.transfer_rate + 0.04 * instability_push)
        aht = max(0.0, 0.20 * prev.aht + 0.80 * current.aht + 12.0 * current.abandon_rate + 10.0 * current.repeat_auth_rate)
        callback = clamp01(0.35 * prev.callback_rate + 0.75 * current.callback_rate)
        backoffice = clamp01(0.30 * prev.backoffice_trigger + 0.75 * current.backoffice_trigger)
        repeat_auth = clamp01(0.30 * prev.repeat_auth_rate + 0.80 * current.repeat_auth_rate)
        short_disconnect = clamp01(0.30 * prev.short_disconnect_rate + 0.80 * current.short_disconnect_rate)
        
        determinism = clamp01(0.20 * prev.determinism_index + 0.80 * current.determinism_index - 0.03 * current.ivr_reentry_rate)
        containment = clamp01(1.0 - safe_rate_sum(blended_abandon, transfer, callback * 0.30))
        
        return Metrics(blended_abandon, blended_reentry, transfer, aht, callback, backoffice, determinism, repeat_auth, containment, short_disconnect)

    def run(self, model: BehaviorModel, horizon: int, summary: CohortSummary, delta_fn: Callable[[int], TouchpointDelta]) -> SimulationResult:
        if horizon <= 0: raise ValueError("horizon must be a positive integer")
        predictions, baseline_energies, step_energies, classes = [], [], [], []
        prev = self.baseline
        
        for t in range(horizon):
            delta = delta_fn(t)
            current = model.generate_projected_metrics(prev, summary, delta)
            
            baseline_energy = Lyapunov.energy(self.baseline, current)
            step_energy = Lyapunov.energy(prev, current)
            state_class = Lyapunov.classify(baseline_energy)
            
            predictions.append(current)
            baseline_energies.append(baseline_energy)
            step_energies.append(step_energy)
            classes.append(state_class)
            
            prev = self.feedback(prev, current)
            
        return SimulationResult(predictions, baseline_energies, step_energies, classes)# ================================# Dashboard and Exports# ================================class Dashboard:    @staticmethod
